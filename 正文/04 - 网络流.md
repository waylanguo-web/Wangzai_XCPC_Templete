## 第四章：网络流

### 最大流

> **用途：** 用 Dinic 求指定源汇之间的最大流。
>
> **复杂度：** $O(N^2M)$ 最坏时间，$O(N+M)$ 空间。
>
> **使用条件：** 容量非负，Flow(n) 可使用 0 至 n。work 会修改残量，重新计算前调用 reset。

使用 $\tt Dinic$ 算法，理论最坏复杂度为 $\mathcal O(N^2M)$ ，例题范围：$N=1200,\ m=5\times 10^3$ 。一般步骤为 BFS 建立分层图，再由 DFS 和当前弧求阻塞流。封装：求从点 $S$ 到点 $T$ 的最大流。预流推进见==常数优化章节==。

| 操作 | 含义 | 复杂度 | 参数范围 |
| --- | --- | --- | --- |
| `add(u,v,c)` | 添加有向容量边 | 摊还 $O(1)$ | 非负容量，允许重边 |
| `work(s,t)` | 在当前残量网络上增广 | $O(N^2M)$ | 源汇不同，返回新增流量 |
| `reset()` | 恢复所有原始边容量 | $O(M)$ | 可用于多源汇重复求割 |

```c++
template<typename T> struct Flow_ {
    const int n;
    const T inf = numeric_limits<T>::max();
    struct Edge {
        int to;
        T w;
        Edge(int to, T w) : to(to), w(w) {}
    };
    vector<Edge> ver;
    vector<vector<int>> h;
    vector<int> cur, d;
    
    Flow_(int n) : n(n + 1), h(n + 1) {}
    void add(int u, int v, T c) {
        assert(c >= 0);
        h[u].push_back(ver.size());
        ver.emplace_back(v, c);
        h[v].push_back(ver.size());
        ver.emplace_back(u, 0);
    }
    bool bfs(int s, int t) {
        d.assign(n, -1);
        d[s] = 0;
        queue<int> q;
        q.push(s);
        while (!q.empty()) {
            auto x = q.front();
            q.pop();
            for (auto it : h[x]) {
                auto yData = ver[it];
                auto y = yData.to;
                auto w = yData.w;
                if (w && d[y] == -1) {
                    d[y] = d[x] + 1;
                    if (y == t) return true;
                    q.push(y);
                }
            }
        }
        return false;
    }
    T dfs(int u, int t, T f) {
        if (u == t) return f;
        auto r = f;
        for (int &i = cur[u]; i < (int)h[u].size(); i++) {
            auto j = h[u][i];
            auto &vData = ver[j];
            auto &v = vData.to;
            auto &c = vData.w;
            auto &rc = ver[j ^ 1].w;
            if (c && d[v] == d[u] + 1) {
                auto a = dfs(v, t, std::min(r, c));
                c -= a;
                rc += a;
                r -= a;
                if (!r) return f;
            }
        }
        return f - r;
    }
    void reset() {
        for (int i = 0; i < (int)ver.size(); i += 2) {
            ver[i].w += ver[i ^ 1].w;
            ver[i ^ 1].w = 0;
        }
    }
    T work(int s, int t) {
        T ans = 0;
        while (bfs(s, t)) {
            cur.assign(n, 0);
            ans += dfs(s, t, inf);
        }
        return ans;
    }
};
using Flow = Flow_<int>;
```

### 最小割

基础模型：输入 n 个材料的非负成本和 m 个项目。每个项目给出两个所需材料的编号 x、y 及非负利润 w。多个项目可以共用已购买的材料。

从源点连边到项目（容量为利润），从材料连边到汇点（容量为成本），从项目连边到依赖的材料（容量大于总利润）。最大净收益为总利润减去最小割。要求总利润小于有符号 64 位上界。

```c++
signed main() {
    int n, m;
    cin >> n >> m;
    const int s = n + m + 1, t = s + 1;
    Flow flow(t);
    for (int i = 1; i <= n; ++i) {
        int cost;
        cin >> cost;
        flow.add(i, t, cost);
    }
    vector<array<int, 3>> projects(m);
    int profit = 0;
    for (auto &item : projects) {
        auto &x = get<0>(item);
        auto &y = get<1>(item);
        auto &w = get<2>(item);
        cin >> x >> y >> w;
        profit += w;
    }
    for (int j = 0; j < m; ++j) {
        auto xData = projects[j];
        auto x = get<0>(xData);
        auto y = get<1>(xData);
        auto w = get<2>(xData);
        int project = n + j + 1;
        flow.add(s, project, w);
        flow.add(project, x, profit + 1);
        flow.add(project, y, profit + 1);
    }
    cout << profit - flow.work(s, t) << '\n';
}
```

### 最小割树 Gomory-Hu Tree

> **用途：** 构造无向图的最小割树，两点最小割等于树上路径的最小边权。
>
> **复杂度：** 构造调用 $N-1$ 次 Dinic，最坏时间 $O(N^3M)$。示例预处理所有询问用时和空间均为 $O(N^2)$。
>
> **使用条件：** 前置依赖为本章的 Flow。顶点编号为 $1$ 至 $N$，容量非负且最大流能存入有符号 64 位整数。询问两点互异。

```c++
vector<vector<pair<int, int>>> gomoryHu(Flow &flow, int n) {
    vector<int> parent(n + 1, 1), weight(n + 1);
    for (int s = 2; s <= n; ++s) {
        int t = parent[s];
        flow.reset();
        weight[s] = flow.work(s, t);
        vector<bool> reachable(n + 1);
        queue<int> q;
        reachable[s] = true;
        q.push(s);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int id : flow.h[u]) {
                auto vData = flow.ver[id];
                auto v = vData.to;
                auto cap = vData.w;
                if (!cap || reachable[v]) continue;
                reachable[v] = true;
                q.push(v);
            }
        }
        for (int v = s + 1; v <= n; ++v) {
            if (parent[v] == t && reachable[v]) parent[v] = s;
        }
        if (t != 1 && reachable[parent[t]]) {
            parent[s] = parent[t];
            parent[t] = s;
            swap(weight[s], weight[t]);
        }
    }
    flow.reset();
    vector<vector<pair<int, int>>> tree(n + 1);
    for (int s = 2; s <= n; ++s) {
        int t = parent[s];
        tree[s].emplace_back(t, weight[s]);
        tree[t].emplace_back(s, weight[s]);
    }
    return tree;
}

signed main() { // Gomory-Hu 查询示例
    int n, m;
    cin >> n >> m;
    Flow flow(n);
    for (int i = 0; i < m; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        flow.add(u, v, w);
        flow.add(v, u, w);
    }
    auto tree = gomoryHu(flow, n);
    vector<vector<int>> ans(n + 1, vector<int>(n + 1));
    for (int root = 1; root <= n; ++root) {
        auto dfs = [&](auto self, int u, int parent, int cut) -> void {
            ans[root][u] = cut;
            for (auto item : tree[u]) {
                auto v = get<0>(item);
                auto w = get<1>(item);
                if (v != parent) self(self, v, u, min(cut, w));
            }
        };
        dfs(dfs, root, 0, numeric_limits<int>::max());
    }
    int q;
    cin >> q;
    while (q--) {
        int u, v;
        cin >> u >> v;
        cout << ans[u][v] << "\n";
    }
}
```

### 费用流

> **用途：** 用势能与逐次最短增广求最小费用最大流。
>
> **复杂度：** 每次增广 $O(M\log(M+1)+N)$；负费用边存在时初始化 $O(NM)$。
>
> **使用条件：** 点编号为 0 至 n−1；初始残量图不得有负费用环。

给定一个带费用的网络，规定 $(u,v)$ 间的费用为 $f(u,v) \times w(u,v)$ ，求解该网络中总花费最小的最大流称之为**最小费用最大流**。每次增广做一次堆优化最短路，时间 $O(M\log(M+1)+N)$。若初始有负费用边，额外花费 $O(NM)$ 初始化势能。

> **使用条件：** MinCostFlow(n) 的点编号为 $0$ 至 $n-1$，容量非负，初始残量网络不得有负费用环。单位费用及其相反数、势能、距离和返回值须能存入 int。源汇相同返回 {0, 0}。

```c++
struct MinCostFlow {
    using PII = pair<int, int>;
    const int INF = numeric_limits<int>::max();
    struct Edge {
        int v, c, f;
        Edge(int v, int c, int f) : v(v), c(c), f(f) {}
    };
    const int n;
    vector<Edge> e;
    vector<vector<int>> g;
    vector<int> h, dis;
    vector<int> pre;
    
    MinCostFlow(int n) : n(n), g(n) {}
    void add(int u, int v, int c, int f) { // c 流量, f 费用
        assert(c >= 0 && f != numeric_limits<int>::min());
        g[u].push_back(e.size());
        e.emplace_back(v, c, f);
        g[v].push_back(e.size());
        e.emplace_back(u, 0, -f);
    }
    bool dijkstra(int s, int t) {
        dis.assign(n, INF);
        pre.assign(n, -1);
        priority_queue<PII, vector<PII>, greater<PII>> que;
        dis[s] = 0;
        que.emplace(0, s);
        while (!que.empty()) {
            auto dData = que.top();
            auto d = get<0>(dData);
            auto u = get<1>(dData);
            que.pop();
            if (dis[u] < d) continue;
            for (int i : g[u]) {
                auto vData = e[i];
                auto v = vData.v;
                auto c = vData.c;
                auto f = vData.f;
                if (c > 0 && dis[v] > d + h[u] - h[v] + f) {
                    dis[v] = d + h[u] - h[v] + f;
                    pre[v] = i;
                    que.emplace(dis[v], v);
                }
            }
        }
        return dis[t] != INF;
    }
    pair<int, int> flow(int s, int t) {
        int flow = 0;
        if (s == t) return {0, 0};
        __int128 cost = 0;
        h.assign(n, 0);
        // 用虚拟源点计算初始势能，无负费用边时无需更新。
        for (int round = 1; round <= n; ++round) {
            bool changed = false;
            for (int i = 0; i < (int)e.size(); ++i) {
                auto vData = e[i];
                auto v = vData.v;
                auto cap = vData.c;
                auto fee = vData.f;
                int u = e[i ^ 1].v;
                if (cap > 0 && h[v] > h[u] + fee) {
                    h[v] = h[u] + fee;
                    changed = true;
                }
            }
            if (!changed) break;
            assert(round < n); // 第 n 轮仍能松弛，存在负费用环
        }
        while (dijkstra(s, t)) {
            for (int i = 0; i < n; ++i) {
                if (dis[i] != INF) h[i] += dis[i];
            }
            int aug = numeric_limits<int>::max();
            for (int i = t; i != s; i = e[pre[i] ^ 1].v) aug = min(aug, e[pre[i]].c);
            for (int i = t; i != s; i = e[pre[i] ^ 1].v) {
                e[pre[i]].c -= aug;
                e[pre[i] ^ 1].c += aug;
            }
            flow += aug;
            cost += (__int128)aug * ((__int128)h[t] - h[s]);
        }
        assert(cost == static_cast<int>(cost));
        return {flow, static_cast<int>(cost)};
    }
};  
```
